??? problem "LeetCode 15 · 三数之和"
    [打开原题 ↗](https://leetcode.cn/problems/3sum/){ .problem-source }

    **题意**：给定整数数组，找出所有元素和为 $0$ 且互不重复的三元组。

    **思路**：排序后枚举最小元素，用相向双指针寻找另外两个数；根据三数之和移动指针，并在三个位置都跳过重复值。

    **复杂度**：时间 $O(n^2)$，除答案外额外空间 $O(1)$。

    **C++ 实现**

    ```cpp
    #include <bits/stdc++.h>
    using namespace std;
    class Solution {
    public:
      vector<vector<int>> threeSum(vector<int>& nums) {
        sort(nums.begin(), nums.end());
        vector<vector<int>> answer;
        int n = nums.size();
        for (int i = 0; i + 2 < n; ++i) {
          if (i > 0 && nums[i] == nums[i - 1]) continue;
          if (nums[i] > 0) break;
          int left = i + 1, right = n - 1;
          while (left < right) {
            long long sum = (long long)nums[i] + nums[left] + nums[right];
            if (sum < 0) {
              ++left;
            } else if (sum > 0) {
              --right;
            } else {
              answer.push_back({nums[i], nums[left], nums[right]});
              int leftValue = nums[left], rightValue = nums[right];
              while (left < right && nums[left] == leftValue) ++left;
              while (left < right && nums[right] == rightValue) --right;
            }
          }
        }
        return answer;
      }
    };
    ```

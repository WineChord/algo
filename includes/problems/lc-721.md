??? problem "LeetCode 721 · 账户合并"
    [打开原题 ↗](https://leetcode.cn/problems/accounts-merge/){ .problem-source }

    **题意**：若两个账户共享任意邮箱，则属于同一人；合并所有连通账户并按字典序输出邮箱。

    **思路**：记录每个邮箱首次所属账户，用并查集合并共享邮箱的账户；随后按代表元收集并排序邮箱。

    **复杂度**：设邮箱总数为 $E$，时间 $O(E\alpha(n)+E\log E)$，额外空间 $O(E+n)$。

    **C++ 实现**

    ```cpp
    #include <bits/stdc++.h>
    using namespace std;
    class Solution {
        vector<int> p;
        int find(int x) {
            return p[x] == x ? x : p[x] = find(p[x]);
        }
    public:
        vector<vector<string>> accountsMerge(vector<vector<string>>& a) {
            int n = a.size();
            p.resize(n);
            iota(p.begin(), p.end(), 0);
            unordered_map<string, int> owner;
            for (int i = 0; i < n; ++i) {
                for (int j = 1; j < (int)a[i].size(); ++j) {
                    auto [it, fresh] = owner.emplace(a[i][j], i);
                    if (!fresh) p[find(i)] = find(it->second);
                }
            }
            unordered_map<int, vector<string>> groups;
            for (auto& [mail, i] : owner) groups[find(i)].push_back(mail);
            vector<vector<string>> ans;
            for (auto& [root, mails] : groups) {
                sort(mails.begin(), mails.end());
                vector<string> cur{a[root][0]};
                cur.insert(cur.end(), mails.begin(), mails.end());
                ans.push_back(move(cur));
            }
            return ans;
        }
    };
    ```

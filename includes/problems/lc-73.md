??? problem "LeetCode 73 · 矩阵置零"
    [打开原题 ↗](https://leetcode.cn/problems/set-matrix-zeroes/){ .problem-source }

    **难度与分值**：LeetCode 官方「中等」。

    **题意**：若矩阵某元素原本为零，就把其整行整列原地置零。

    **思路**：用首行与首列存其余行列是否应清零，再用两个布尔量单独保存首行首列的原始状态，最后逆序落地。

    **复杂度**：时间 $O(mn)$，额外空间 $O(1)$。

    **C++ 实现**

    <!-- compile:leetcode -->
    ```cpp
    #include <bits/stdc++.h>
    using namespace std;
    class Solution {
    public:
      void setZeroes(vector<vector<int>>& matrix) {
        int m = matrix.size();
        int n = matrix[0].size();
        bool firstRowZero = false;
        bool firstColumnZero = false;
        for (int j = 0; j < n; ++j) {
          firstRowZero |= matrix[0][j] == 0;
        }
        for (int i = 0; i < m; ++i) {
          firstColumnZero |= matrix[i][0] == 0;
        }
        for (int i = 1; i < m; ++i) {
          for (int j = 1; j < n; ++j) {
            if (matrix[i][j] == 0) {
              matrix[i][0] = 0;
              matrix[0][j] = 0;
            }
          }
        }
        for (int i = 1; i < m; ++i) {
          for (int j = 1; j < n; ++j) {
            if (matrix[i][0] == 0 || matrix[0][j] == 0) {
              matrix[i][j] = 0;
            }
          }
        }
        if (firstRowZero) {
          fill(matrix[0].begin(), matrix[0].end(), 0);
        }
        if (firstColumnZero) {
          for (int i = 0; i < m; ++i) {
            matrix[i][0] = 0;
          }
        }
      }
    };
    ```

??? problem "LeetCode 200 · 岛屿数量"
    [打开原题 ↗](https://leetcode.cn/problems/number-of-islands/){ .problem-source }

    **难度**：LeetCode 官方「中等」。

    **题意**：在由陆地 `1` 和水域 `0` 组成的网格中，四方向相连的陆地构成一座岛，求岛屿数量。

    **思路**：逐格扫描；每遇到尚未访问的陆地，答案加一，并用显式栈把整座岛原地改成水域。一次搜索恰好消去一个连通分量，并避开深递归风险。

    **复杂度**：时间 $O(mn)$，显式栈空间最坏 $O(mn)$。

    **C++ 实现**

    ```cpp
    #include <bits/stdc++.h>
    using namespace std;
    class Solution {
    public:
      int numIslands(vector<vector<char>>& grid) {
        int m = grid.size(), n = grid[0].size(), answer = 0;
        static const int dx[4] = {1, -1, 0, 0};
        static const int dy[4] = {0, 0, 1, -1};
        for (int startX = 0; startX < m; ++startX) {
          for (int startY = 0; startY < n; ++startY) {
            if (grid[startX][startY] != '1') continue;
            ++answer;
            stack<pair<int, int>> pending;
            pending.push({startX, startY});
            grid[startX][startY] = '0';
            while (!pending.empty()) {
              auto [x, y] = pending.top();
              pending.pop();
              for (int direction = 0; direction < 4; ++direction) {
                int nextX = x + dx[direction];
                int nextY = y + dy[direction];
                if (nextX < 0 || nextX >= m || nextY < 0 || nextY >= n) continue;
                if (grid[nextX][nextY] != '1') continue;
                grid[nextX][nextY] = '0';
                pending.push({nextX, nextY});
              }
            }
          }
        }
        return answer;
      }
    };
    ```

??? problem "LeetCode 4015 · 树的加权和"
    [打开原题 ↗](https://leetcode.cn/problems/weighted-sum-of-a-tree/){ .problem-source }

    **难度与分值**：LeetCode 官方「中等」，第 514 场周赛官方 4 分；ZeroTracer 社区估算竞赛分未知（核对于 2026-08-15）。

    **题意**：有根树中节点深度从 1 开始，树高为最大深度；节点 i 的权重是 nums[i]·(height-depth[i]+1)，求全部权重之和。

    **思路**：父节点编号未必小于孩子，先建立孩子表，再从根 BFS，一次得到全部深度与树高。第二次线性扫描代入权重公式，并用 64 位整数累加。

    **复杂度**：时间 O(n)，空间 O(n)。

    **C++ 实现**

    <!-- compile:standalone -->
    ```cpp
    #include <bits/stdc++.h>
    using namespace std;
    class Solution {
    public:
      long long weightedSum(vector<int>& parent, vector<int>& nums) {
        int n = static_cast<int>(parent.size());
        vector<vector<int>> children(n);
        for (int node = 1; node < n; ++node) {
          children[parent[node]].push_back(node);
        }
        vector<int> depth(n);
        queue<int> pending;
        depth[0] = 1;
        pending.push(0);
        int height = 1;
        while (!pending.empty()) {
          int node = pending.front();
          pending.pop();
          for (int child : children[node]) {
            depth[child] = depth[node] + 1;
            height = max(height, depth[child]);
            pending.push(child);
          }
        }
        long long answer = 0;
        for (int node = 0; node < n; ++node) {
          answer += 1LL * nums[node] * (height - depth[node] + 1);
        }
        return answer;
      }
    };
    int main() {
      vector<int> parent{-1, 0, 1, 2};
      vector<int> nums{1, 2, 3, 4};
      cout << Solution().weightedSum(parent, nums) << '\n';
    }
    ```

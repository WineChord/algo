??? problem "LeetCode 141 · 环形链表"
    [打开原题 ↗](https://leetcode.cn/problems/linked-list-cycle/){ .problem-source }

    **难度与分值**：LeetCode 官方「简单」。

    **题意**：判断只读单链表沿 `next` 指针是否会重复访问同一节点。

    **思路**：快指针每轮比慢指针多走一步；有环时环上相对距离必归零，无环时快指针先到空。

    **复杂度**：时间 $O(n)$，额外空间 $O(1)$。

    **C++ 实现**

    ```cpp
    #include <bits/stdc++.h>
    using namespace std;
    struct ListNode {
      int val;
      ListNode* next;
      ListNode(int value = 0, ListNode* following = nullptr) : val(value), next(following) {
      }
    };
    class Solution {
    public:
      bool hasCycle(ListNode* head) {
        ListNode* slow = head;
        ListNode* fast = head;
        while (fast && fast->next) {
          slow = slow->next;
          fast = fast->next->next;
          if (slow == fast) {
            return true;
          }
        }
        return false;
      }
    };
    ```

??? problem "LeetCode 279 · 完全平方数"
    [打开原题 ↗](https://leetcode.cn/problems/perfect-squares/){ .problem-source }

    **难度与分值**：LeetCode 官方「中等」。

    **题意**：给定正整数 $n$，求其表示成完全平方数之和所需的最少项数。

    **思路**：先判一平方；由勒让德三平方定理判答案是否为四；再枚举一个平方判是否为二，余下情况必为三。

    **复杂度**：时间 $O(\sqrt n)$，额外空间 $O(1)$。

    **C++ 实现**

    ```cpp
    #include <bits/stdc++.h>
    using namespace std;
    class Solution {
      bool isSquare(int value) {
        int root = static_cast<int>(sqrt(value));
        while (1LL * root * root < value) {
          ++root;
        }
        while (1LL * root * root > value) {
          --root;
        }
        return root * root == value;
      }
    public:
      int numSquares(int n) {
        if (isSquare(n)) {
          return 1;
        }
        int reduced = n;
        while (reduced % 4 == 0) {
          reduced /= 4;
        }
        if (reduced % 8 == 7) {
          return 4;
        }
        for (int root = 1; root * root <= n; ++root) {
          if (isSquare(n - root * root)) {
            return 2;
          }
        }
        return 3;
      }
    };
    ```

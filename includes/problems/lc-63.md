??? problem "LeetCode 63 · 不同路径 II"
    [打开原题 ↗](https://leetcode.cn/problems/unique-paths-ii/){ .problem-source }

    **难度与分值**：LeetCode 官方「中等」。

    **题意**：只能向右或向下移动的网格中存在障碍，统计从左上角到右下角的不同路径数。

    **思路**：按行扫描，`dp[j]` 更新前代表上方路径数、`dp[j-1]` 更新后代表左方路径数；障碍格直接把状态清零。

    **复杂度**：时间 $O(mn)$，空间 $O(n)$。

    **C++ 实现**

    <!-- compile:standalone -->
    ```cpp
    #include <bits/stdc++.h>
    using namespace std;
    class Solution {
    public:
      int uniquePathsWithObstacles(vector<vector<int>>& grid) {
        constexpr long long CAP = 2'000'000'001LL;
        int rows = grid.size(), columns = grid[0].size();
        vector<long long> dp(columns);
        dp[0] = grid[0][0] == 0;
        for (int row = 0; row < rows; ++row) {
          for (int column = 0; column < columns; ++column) {
            if (grid[row][column]) {
              dp[column] = 0;
            } else if (column > 0) {
              dp[column] = min(CAP, dp[column] + dp[column - 1]);
            }
          }
        }
        return static_cast<int>(dp.back());
      }
    };
    int main() {
      vector<vector<int>> grid{{0, 0, 0}, {0, 1, 0}, {0, 0, 0}};
      cout << Solution().uniquePathsWithObstacles(grid) << '\n';
    }
    ```

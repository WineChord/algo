??? problem "LeetCode 3518 · 最小回文排列 II"
    [打开原题 ↗](https://leetcode.cn/problems/smallest-palindromic-rearrangement-ii/){ .problem-source }

    **难度与分值**：LeetCode 官方「困难」；[ZeroTracer](https://zerotrac.github.io/leetcode_problem_rating/) 社区估算竞赛分为 2375（检索于 2026-07-29）。

    **题意**：把给定回文串的字符重新排列，返回字典序第 $k$ 小的不同回文排列；不足 $k$ 个时返回空串。

    **思路**：回文由半串唯一决定。从左到右尝试字符，用截断到 $k$ 的多重集排列数计算每个字典序块大小，从而整块跳过而不枚举排列。

    **复杂度**：保守时间 $O(26^2n\log k)$，实际因计数截断接近 $O(26n\log k)$；额外空间 $O(26+n)$。

    **C++ 实现**

    ```cpp
    #include <bits/stdc++.h>
    using namespace std;
    class Solution {
      int combinationCapped(int n, int r, int limit) {
        r = min(r, n - r);
        long long value = 1;
        for (int i = 1; i <= r; ++i) {
          __int128 next = static_cast<__int128>(value) * (n - r + i);
          next /= i;
          if (next >= limit) {
            return limit;
          }
          value = static_cast<long long>(next);
        }
        return static_cast<int>(value);
      }
      int permutationsCapped(const array<int, 26>& count, int limit) {
        long long total = 1;
        int used = 0;
        for (int amount : count) {
          if (amount == 0) {
            continue;
          }
          int required = static_cast<int>((limit + total - 1) / total);
          int ways = combinationCapped(used + amount, amount, required);
          if (ways >= required) {
            return limit;
          }
          total *= ways;
          used += amount;
        }
        return static_cast<int>(total);
      }
    public:
      string smallestPalindrome(string s, int k) {
        array<int, 26> full{};
        for (char ch : s) {
          ++full[ch - 'a'];
        }
        array<int, 26> halfCount{};
        char middle = '\0';
        int halfLength = 0;
        for (int ch = 0; ch < 26; ++ch) {
          halfCount[ch] = full[ch] / 2;
          halfLength += halfCount[ch];
          if (full[ch] % 2 == 1) {
            middle = static_cast<char>('a' + ch);
          }
        }
        if (permutationsCapped(halfCount, k) < k) {
          return "";
        }
        string left;
        left.reserve(halfLength);
        for (int position = 0; position < halfLength; ++position) {
          for (int ch = 0; ch < 26; ++ch) {
            if (halfCount[ch] == 0) {
              continue;
            }
            --halfCount[ch];
            int ways = permutationsCapped(halfCount, k);
            if (ways >= k) {
              left.push_back(static_cast<char>('a' + ch));
              break;
            }
            k -= ways;
            ++halfCount[ch];
          }
        }
        string right = left;
        reverse(right.begin(), right.end());
        if (middle == '\0') {
          return left + right;
        }
        return left + middle + right;
      }
    };
    ```

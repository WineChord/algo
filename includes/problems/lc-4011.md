??? problem "LeetCode 4011 · 按奇偶比统计子数组 I"
    [打开原题 ↗](https://leetcode.cn/problems/count-subarrays-with-even-odd-ratio-i/){ .problem-source }

    **难度与分值**：LeetCode 官方「中等」，第 513 场周赛官方 4 分；ZeroTracer 社区估算竞赛分未知（检索于 2026-08-04）。

    **题意**：统计奇数个数 $y>0$ 且偶数个数与奇数个数满足 $x/y\le a/b$ 的非空连续子数组。

    **思路**：把偶数记为 $+b$、奇数记为 $-a$，比例约束化为左右前缀序关系；按右端扫描，用 Fenwick 树统计不小于当前值的历史前缀。

    **复杂度**：时间 $O(n\log n)$，空间 $O(n)$；在本题规模下也可用 $O(n^2)$ 枚举。

    **C++ 实现**

    ```cpp
    #include <bits/stdc++.h>
    using namespace std;
    class Solution {
      class Fenwick {
        vector<int> tree;
      public:
        explicit Fenwick(int size) : tree(size + 1) {
        }
        void add(int index) {
          for (++index; index < static_cast<int>(tree.size()); index += index & -index) {
            ++tree[index];
          }
        }
        int prefix(int index) const {
          int answer = 0;
          for (++index; index > 0; index -= index & -index) {
            answer += tree[index];
          }
          return answer;
        }
      };
    public:
      int countRatioSubarrays(vector<int>& nums, int a, int b) {
        vector<int> prefix(1, 0);
        for (int value : nums) {
          int weight = value % 2 == 0 ? b : -a;
          prefix.push_back(prefix.back() + weight);
        }
        vector<int> coordinates = prefix;
        sort(coordinates.begin(), coordinates.end());
        coordinates.erase(unique(coordinates.begin(), coordinates.end()), coordinates.end());
        Fenwick fenwick(coordinates.size());
        int seen = 0;
        int answer = 0;
        for (int value : prefix) {
          int rank = lower_bound(coordinates.begin(), coordinates.end(), value) - coordinates.begin();
          answer += seen - fenwick.prefix(rank - 1);
          fenwick.add(rank);
          ++seen;
        }
        return answer;
      }
    };
    ```

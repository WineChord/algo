??? problem "LeetCode 912 · 排序数组"
    [打开原题 ↗](https://leetcode.cn/problems/sort-an-array/){ .problem-source }

    **难度与分值**：LeetCode 官方「中等」。

    **题意**：在不调用内置排序函数的前提下把整数数组升序排列，并达到 $O(n\log n)$ 时间。

    **思路**：原地建最大堆，反复把堆顶最大值换到未排序区间末尾，再向下调整剩余堆。

    **复杂度**：时间 $O(n\log n)$，额外空间 $O(1)$。

    **C++ 实现**

    ```cpp
    #include <bits/stdc++.h>
    using namespace std;
    class Solution {
      void siftDown(vector<int>& a, int root, int size) {
        while (2 * root + 1 < size) {
          int child = 2 * root + 1;
          if (child + 1 < size && a[child] < a[child + 1]) {
            ++child;
          }
          if (a[root] >= a[child]) {
            return;
          }
          swap(a[root], a[child]);
          root = child;
        }
      }
    public:
      vector<int> sortArray(vector<int>& nums) {
        int n = nums.size();
        for (int i = n / 2 - 1; i >= 0; --i) {
          siftDown(nums, i, n);
        }
        for (int size = n; size > 1; --size) {
          swap(nums[0], nums[size - 1]);
          siftDown(nums, 0, size - 1);
        }
        return nums;
      }
    };
    ```

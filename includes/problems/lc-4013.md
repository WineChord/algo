??? problem "LeetCode 4013 · 按奇偶比统计子数组 II"
    [打开原题 ↗](https://leetcode.cn/problems/count-subarrays-with-even-odd-ratio-ii/){ .problem-source }

    **难度与分值**：LeetCode 官方「困难」，第 513 场周赛官方 6 分；ZeroTracer 社区估算竞赛分未知（检索于 2026-08-06）。

    **题意**：统计偶数个数与奇数个数满足指定比例条件的非空子数组。

    **思路**：把比例不等式化为前缀势能 $V_i=bE_i-aO_i$；答案是满足 $l<r$ 且 $V_l\ge V_r$ 的前缀对，用坐标压缩 Fenwick 树在线计数。

    **复杂度**：时间 $O(n\log n)$，额外空间 $O(n)$。

    **C++ 实现**

    <!-- compile:leetcode -->
    ```cpp
    #include <bits/stdc++.h>
    using namespace std;
    class Fenwick {
      vector<int> tree;
    public:
      explicit Fenwick(int n) : tree(n + 1) {
      }
      void add(int index) {
        for (; index < static_cast<int>(tree.size()); index += index & -index) {
          ++tree[index];
        }
      }
      int sum(int index) const {
        int answer = 0;
        for (; index > 0; index -= index & -index) {
          answer += tree[index];
        }
        return answer;
      }
    };
    class Solution {
    public:
      long long countRatioSubarrays(vector<int>& nums, int a, int b) {
        int n = nums.size();
        vector<long long> value(n + 1);
        long long even = 0;
        long long odd = 0;
        for (int i = 0; i < n; ++i) {
          even += nums[i] % 2 == 0;
          odd += nums[i] % 2 != 0;
          value[i + 1] = even * b - odd * a;
        }
        vector<long long> order = value;
        sort(order.begin(), order.end());
        order.erase(unique(order.begin(), order.end()), order.end());
        Fenwick fenwick(order.size());
        long long answer = 0;
        for (int i = 0; i <= n; ++i) {
          int rank = lower_bound(order.begin(), order.end(), value[i]) - order.begin() + 1;
          answer += i - fenwick.sum(rank - 1);
          fenwick.add(rank);
        }
        return answer;
      }
    };
    ```

??? problem "LeetCode 51 · N 皇后"
    [打开原题 ↗](https://leetcode.cn/problems/n-queens/){ .problem-source }

    **难度与分值**：LeetCode 官方「困难」。

    **题意**：返回在 $n\times n$ 棋盘放置 $n$ 个互不攻击皇后的全部布局。

    **思路**：逐行放置皇后，并用列、主对角线和副对角线三个掩码保存攻击线。每层只枚举三类掩码均未占用的列，从而常数时间判定冲突。

    **复杂度**：搜索规模最坏 $O(n!)$，递归额外空间 $O(n)$。

    **C++ 实现**

    ```cpp
    #include <bits/stdc++.h>
    using namespace std;
    class Solution {
      vector<vector<string>> answer;
      vector<int> queenColumn;
      int fullMask;
      void search(int row, int columns, int diagonalLeft, int diagonalRight) {
        int n = queenColumn.size();
        if (row == n) {
          vector<string> board(n, string(n, '.'));
          for (int r = 0; r < n; ++r) {
            board[r][queenColumn[r]] = 'Q';
          }
          answer.push_back(board);
          return;
        }
        int available = fullMask & ~(columns | diagonalLeft | diagonalRight);
        while (available) {
          int position = available & -available;
          available -= position;
          queenColumn[row] = countr_zero(static_cast<unsigned int>(position));
          search(row + 1, columns | position, ((diagonalLeft | position) << 1) & fullMask,
              (diagonalRight | position) >> 1);
        }
      }
    public:
      vector<vector<string>> solveNQueens(int n) {
        queenColumn.resize(n);
        fullMask = (1 << n) - 1;
        search(0, 0, 0, 0);
        return answer;
      }
    };
    ```

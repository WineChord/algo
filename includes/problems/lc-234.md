??? problem "LeetCode 234 · 回文链表"
    [打开原题 ↗](https://leetcode.cn/problems/palindrome-linked-list/){ .problem-source }

    **难度与分值**：LeetCode 官方「简单」。

    **题意**：判断单链表的节点值序列是否为回文。

    **思路**：快慢指针找到后半段，原地反转后与前半段同步比较，最后再次反转恢复输入结构。

    **复杂度**：时间 $O(n)$，额外空间 $O(1)$。

    **C++ 实现**

    ```cpp
    #include <bits/stdc++.h>
    using namespace std;
    struct ListNode {
      int val = 0;
      ListNode* next = nullptr;
    };
    class Solution {
      ListNode* reverseList(ListNode* head) {
        ListNode* previous = nullptr;
        while (head) {
          ListNode* next = head->next;
          head->next = previous;
          previous = head;
          head = next;
        }
        return previous;
      }
    public:
      bool isPalindrome(ListNode* head) {
        ListNode* slow = head;
        ListNode* fast = head;
        while (fast->next && fast->next->next) {
          slow = slow->next;
          fast = fast->next->next;
        }
        ListNode* reversed = reverseList(slow->next);
        ListNode* first = head;
        ListNode* second = reversed;
        bool answer = true;
        while (second) {
          if (first->val != second->val) {
            answer = false;
            break;
          }
          first = first->next;
          second = second->next;
        }
        slow->next = reverseList(reversed);
        return answer;
      }
    };
    ```

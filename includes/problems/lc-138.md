??? problem "LeetCode 138 · 随机链表的复制"
    [打开原题 ↗](https://leetcode.cn/problems/copy-list-with-random-pointer/){ .problem-source }

    **难度与分值**：LeetCode 官方「中等」。

    **题意**：深拷贝带 `next` 与任意 `random` 指针的链表。

    **思路**：把每个副本暂时插在原节点后，使 `original->next` 充当原节点到副本的映射；连接随机边后再拆分并恢复原链。

    **复杂度**：时间 $O(n)$，除返回节点外额外空间 $O(1)$。

    **C++ 实现**

    ```cpp
    #include <bits/stdc++.h>
    using namespace std;
    class Node {
    public:
      int val;
      Node* next;
      Node* random;
      explicit Node(int value) : val(value), next(nullptr), random(nullptr) {
      }
    };
    class Solution {
    public:
      Node* copyRandomList(Node* head) {
        if (!head) {
          return nullptr;
        }
        for (Node* node = head; node;) {
          Node* next = node->next;
          node->next = new Node(node->val);
          node->next->next = next;
          node = next;
        }
        for (Node* node = head; node; node = node->next->next) {
          node->next->random = node->random ? node->random->next : nullptr;
        }
        Node* copiedHead = head->next;
        for (Node* node = head; node;) {
          Node* copy = node->next;
          Node* next = copy->next;
          node->next = next;
          copy->next = next ? next->next : nullptr;
          node = next;
        }
        return copiedHead;
      }
    };
    ```

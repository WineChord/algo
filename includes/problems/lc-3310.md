??? problem "LeetCode 3310 · 移除可疑的方法"
    [打开原题 ↗](https://leetcode.cn/problems/remove-methods-from-project/){ .problem-source }

    **难度与分值**：LeetCode 官方「中等」，2026-08-05 力扣中国每日一题；ZeroTracer 社区估算竞赛分为 1710.52（非官方，检索于 2026-08-05）。

    **题意**：从含 bug 的方法 $k$ 出发，其所有可达方法均可疑；仅当不存在非可疑方法调用可疑方法时，移除全部可疑方法，否则一个也不能移除。

    **思路**：先沿调用边标记从 $k$ 可达的闭包，再扫描所有边寻找从闭包外进入闭包的边；存在则返回全部方法，否则返回闭包补集。

    **复杂度**：时间 $O(n+m)$，额外空间 $O(n+m)$。

    **C++ 实现**

    ```cpp
    #include <bits/stdc++.h>
    using namespace std;
    class Solution {
    public:
      vector<int> remainingMethods(int n, int k, vector<vector<int>>& invocations) {
        vector<vector<int>> graph(n);
        for (const auto& edge : invocations) {
          graph[edge[0]].push_back(edge[1]);
        }
        vector<char> suspicious(n);
        queue<int> pending;
        suspicious[k] = true;
        pending.push(k);
        while (!pending.empty()) {
          int node = pending.front();
          pending.pop();
          for (int next : graph[node]) {
            if (!suspicious[next]) {
              suspicious[next] = true;
              pending.push(next);
            }
          }
        }
        for (const auto& edge : invocations) {
          if (!suspicious[edge[0]] && suspicious[edge[1]]) {
            vector<int> all(n);
            iota(all.begin(), all.end(), 0);
            return all;
          }
        }
        vector<int> answer;
        for (int node = 0; node < n; ++node) {
          if (!suspicious[node]) {
            answer.push_back(node);
          }
        }
        return answer;
      }
    };
    ```

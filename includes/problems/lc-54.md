??? problem "LeetCode 54 · 螺旋矩阵"
    [打开原题 ↗](https://leetcode.cn/problems/spiral-matrix/){ .problem-source }

    **难度**：LeetCode 官方「中等」。

    **题意**：从左上角开始，逐层向内按顺时针螺旋顺序返回矩阵全部元素。

    **思路**：维护尚未访问矩形的上下左右边界。输出一条边后立即收缩；访问下边和左边前再次判断剩余行列，避免单行、单列重复。

    **复杂度**：时间 $O(mn)$，输出之外额外空间 $O(1)$。

    **C++ 实现**

    ```cpp
    #include <bits/stdc++.h>
    using namespace std;
    class Solution {
    public:
      vector<int> spiralOrder(vector<vector<int>>& matrix) {
        int top = 0;
        int bottom = matrix.size() - 1;
        int left = 0;
        int right = matrix[0].size() - 1;
        vector<int> answer;
        answer.reserve(matrix.size() * matrix[0].size());
        while (top <= bottom && left <= right) {
          for (int column = left; column <= right; ++column) answer.push_back(matrix[top][column]);
          ++top;
          for (int row = top; row <= bottom; ++row) answer.push_back(matrix[row][right]);
          --right;
          if (top <= bottom) {
            for (int column = right; column >= left; --column) answer.push_back(matrix[bottom][column]);
            --bottom;
          }
          if (left <= right) {
            for (int row = bottom; row >= top; --row) answer.push_back(matrix[row][left]);
            ++left;
          }
        }
        return answer;
      }
    };
    ```
